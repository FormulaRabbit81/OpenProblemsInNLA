def sha(data):
    return hashlib.sha256(data).hexdigest()

def digest(path):
    return sha(path.read_bytes())

def write_json(name, obj):
    (P / name).write_text(json.dumps(obj, indent=2, ensure_ascii=False) + '\n')

def copy(src, rel):
    dest = P / rel
    dest.parent.mkdir(parents=True, exist_ok=True)
    if dest.exists():
        assert dest.read_bytes() == src.read_bytes(), 'existing reference differs: ' + rel
    else:
        shutil.copyfile(src, dest)
    return {'source': str(src), 'packet_path': rel,
            'sha256': digest(dest), 'bytes': dest.stat().st_size}
