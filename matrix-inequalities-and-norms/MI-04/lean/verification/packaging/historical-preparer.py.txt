def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def read_json(path):
    return json.loads(Path(path).read_text())


def safe_child(root, relative):
    rel = PurePosixPath(relative)
    require(not rel.is_absolute() and '..' not in rel.parts, f'Unsafe path: {relative}')
    path = Path(root) / relative
    require(not path.is_symlink(), f'Symlink is not a retained input: {path}')
    return path


def checked_files(root, entries):
    require(isinstance(entries, dict), 'Expected a path/hash map')
    for rel, digest in entries.items():
        require(re.fullmatch(r'[0-9a-f]{64}', digest) is not None, f'Invalid digest: {rel}')
        require(sha(safe_child(root, rel)) == digest, f'Hash mismatch: {root}/{rel}')


def declaration(text, short_name):
    pattern = re.compile(
        r'^\s*(?:theorem|lemma)\s+' + re.escape(short_name) + r'\b[\s\S]*?\s:=',
        re.MULTILINE,
    )
    matches = list(pattern.finditer(text))
    require(len(matches) == 1, f'Expected one declaration of {short_name}')
    match = matches[0]
    header = match.group(0).strip()
    offset = match.start() + len(match.group(0)) - len(match.group(0).lstrip())
    return header, text.count('\n', 0, offset) + 1


def cosmetic_header(text):
    # Matches the two independent source reviews: lemma/theorem spelling and
    # intentionally unused explicit hypothesis names do not change the type.
    text = re.sub(r'^(theorem|lemma)\s+', '', text)
    text = re.sub(r'\b_(h[A-Za-z0-9_]*)\b', r'\1', text)
    return ''.join(text.split())
