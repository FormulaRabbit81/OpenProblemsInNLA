    worstGMRES (jordan n λ) k = idealGMRES (jordan n λ) k ∧
    (∀ x ∈ unitSphere n, ∃ p : Poly, Admissible k p ∧
      ‖euclideanLin (polyEval p (jordan n λ)) x‖ = gmresInner (jordan n λ) k x ∧
      ∀ q : Poly, Admissible k q → ‖euclideanLin (polyEval p (jordan n λ)) x‖ ≤
        ‖euclideanLin (polyEval q (jordan n λ)) x‖) ∧
    ∃ (p : Poly) (x : H n), Admissible k p ∧ x ∈ unitSphere n ∧
      operatorNorm (polyEval p (jordan n λ)) = idealGMRES (jordan n λ) k ∧
      ‖euclideanLin (polyEval p (jordan n λ)) x‖ = operatorNorm (polyEval p (jordan n λ)) ∧

      (∀ q : Poly, Admissible k q →

      (∀ q : Poly, Admissible k q → ‖euclideanLin (polyEval p (jordan n λ)) x‖ ≤
        ‖euclideanLin (polyEval q (jordan n λ)) x‖) ∧

end
