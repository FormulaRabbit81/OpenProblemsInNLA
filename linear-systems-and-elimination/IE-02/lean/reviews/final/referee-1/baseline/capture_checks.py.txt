def lean_code(text):
    """Remove nested comments/string contents, preserving newlines for a static scan."""
    out = []
    i, depth, string = 0, 0, False
    while i < len(text):
        c = text[i]
        pair = text[i:i+2]
        if depth:
            if pair == '/-':
                depth += 1
                out.extend('  ')
                i += 2
            elif pair == '-/':
                depth -= 1
                out.extend('  ')
                i += 2
            else:
                out.append('\n' if c == '\n' else ' ')
                i += 1
        elif string:
            if c == '\\' and i+1 < len(text):
                out.extend('  ')
                i += 2
            else:
                out.append('\n' if c == '\n' else ' ')
                string = c != '"'
                i += 1
        elif pair == '/-':
            depth = 1
            out.extend('  ')
            i += 2
        elif pair == '--':
            j = text.find('\n', i)
            if j == -1:
                j = len(text)
            out.extend(' ' * (j-i))
            i = j
        elif c == '"':
            string = True
            out.append(' ')
            i += 1
        else:
            out.append(c)
            i += 1
    assert depth == 0 and not string
    return ''.join(out)
