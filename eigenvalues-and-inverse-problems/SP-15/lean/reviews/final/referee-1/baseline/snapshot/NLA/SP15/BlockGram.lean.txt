Derive the Gram matrix by block multiplication, then transport it through

import NLA.SP15.SquareRoots
import Mathlib.LinearAlgebra.Matrix.Reindex
import Mathlib.Tactic.Abel

set_option autoImplicit false
set_option leancert.trust "kernel"

namespace NLA.SP15
noncomputable section
open scoped Matrix

private theorem shifted_gram_expansion {ι : Type*} [Fintype ι] [DecidableEq ι]
    (A : Matrix ι ι ℂ) (z a : ℂ) :
    a • 1 + (A - z • 1).conjTranspose * (A - z • 1) =
      (a + (Complex.normSq z : ℂ)) • 1 + A.conjTranspose * A -
        z • A.conjTranspose - star z • A := by
  have hz : z * star z = (Complex.normSq z : ℂ) := Complex.mul_conj z
  simp only [Matrix.conjTranspose_sub, Matrix.conjTranspose_smul,
    Matrix.conjTranspose_one, sub_mul, mul_sub, Matrix.smul_mul,
    Matrix.mul_smul, Matrix.one_mul, Matrix.mul_one, smul_smul, hz, add_smul]
  abel

private theorem block_gram_diagonal (x : Parameters) (hx : x ∈ parameterBox) :
    (blockMatrix x).conjTranspose * blockMatrix x =
      Matrix.fromBlocks (0 : Square 3)
        (0 : Matrix (Fin 3) (Fin 3 ⊕ Fin 3) ℂ)
        (0 : Matrix (Fin 3 ⊕ Fin 3) (Fin 3) ℂ)
        (Matrix.fromBlocks (pMatrix x) (0 : Square 3) (0 : Square 3)
          ((qRoot x).conjTranspose * qRoot x)) := by
  have hr := square_root_semantics x hx
  have hgram : (pRoot x).conjTranspose * pRoot x = pMatrix x := by
    rw [hr.1.eq, hr.2.2.1]
  simp only [blockMatrix, Matrix.fromRows_zero, Matrix.fromBlocks_conjTranspose,
    Matrix.conjTranspose_fromCols_eq_fromRows_conjTranspose,
    Matrix.conjTranspose_zero, Matrix.fromBlocks_multiply,
    Matrix.fromRows_mul_fromCols,
    Matrix.zero_mul, Matrix.mul_zero, zero_add, add_zero,
    Matrix.fromBlocks_add, hgram]

private theorem regularized_gram_before_reindex (x : Parameters) (hx : x ∈ parameterBox)
    (z : ℂ) (t : ℝ) :
    regularizedGramBlocks x z t =
      (t : ℂ) • (1 : BlockSquare) +
        (blockMatrix x - z • 1).conjTranspose * (blockMatrix x - z • 1) := by
  rw [shifted_gram_expansion, block_gram_diagonal x hx]
  ext i j
  rcases i with i | (i | i) <;> rcases j with j | (j | j) <;>
    simp [regularizedGramBlocks, blockMatrix, shiftScalar, Matrix.conjTranspose_apply,
      Matrix.one_apply, sub_eq_add_neg]

theorem regularized_gram_blocks (x : Parameters) (hx : x ∈ parameterBox)
    (z : ℂ) (t : ℝ) :
    Matrix.reindex blockEquiv blockEquiv (regularizedGramBlocks x z t) =
      (t : ℂ) • (1 : Square 9) + shiftedGram (constructedMatrix x) z := by
  let e := Matrix.reindexAlgEquiv ℂ ℂ blockEquiv
  change e (regularizedGramBlocks x z t) = _
  rw [regularized_gram_before_reindex x hx z t, map_add, map_smul, map_one, map_mul]
  have hstar : e ((blockMatrix x - z • 1).conjTranspose) =
      (e (blockMatrix x - z • 1)).conjTranspose :=
    (Matrix.conjTranspose_reindex blockEquiv blockEquiv (blockMatrix x - z • 1)).symm
  rw [hstar, map_sub, map_smul, map_one]
  rfl

#print axioms regularized_gram_blocks
#assert_trust kernel regularized_gram_blocks

end
end NLA.SP15
