from pathlib import Path
import hashlib
import json
import re

R = Path(__file__).resolve().parent
B = Path('/private/tmp/nla-lean-next-20260915')
P = B / 'next-proofs/SP-15'
L = Path('/private/tmp/nla-lean-local-shared-20260916')
M = L / '.lake/packages/mathlib'

def sha(data):
    return hashlib.sha256(data).hexdigest()

def keep(src, dest):
    data = src.read_bytes()
    out = R / dest
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_bytes(data)
    return {'original': str(src), 'snapshot': str(dest), 'sha256': sha(data), 'bytes': len(data)}

index = json.loads((R / 'SOURCE-SNAPSHOT.json').read_text())
sources = {f['path']: f for f in index['files']}
freeze = json.loads((P / 'STATEMENT-FREEZE.json').read_text())
for rel in freeze['frozen_files']:
    if rel not in sources:
        dest = Path('snapshot') / rel
        if dest.suffix == '.lean':
            dest = dest.with_suffix('.lean.txt')
        rec = keep(P / rel, dest)
        rec.update(path=rel, solution_closure=False)
        sources[rel] = rec
index['files'] = sorted(sources.values(), key=lambda f: f['path'])
(R / 'SOURCE-SNAPSHOT.json').write_text(json.dumps(index, indent=2) + '\n')

records = []
records.append(keep(B / 'SP15-ROOT-FREEZE-ACCEPTANCE.json', 'evidence/SP15-ROOT-FREEZE-ACCEPTANCE.json'))
for rel in ['ASSEMBLY-116.json', 'LOCAL-ENVIRONMENT.json', 'lake-manifest.json', 'lean-toolchain', 'serial_compile_v3.py']:
    dest = Path('evidence/local-environment') / rel
    if dest.suffix == '.py':
        dest = dest.with_suffix('.py.txt')
    records.append(keep(L / rel, dest))

receipt_records = {}
successes = {}
chains = {}
run116 = L / 'runs/development-116/RECEIPT.json'
root_receipt = json.loads(run116.read_text())
latest = {c['module']: c for c in root_receipt['commands']}

def receipt(path):
    if str(path) not in receipt_records:
        receipt_records[str(path)] = keep(path, Path('evidence/local-runs') / path.parent.name / 'RECEIPT.json')
    return json.loads(path.read_text())

def follow(module, path):
    chain = []
    while True:
        data = receipt(path)
        c = next(c for c in data['commands'] if c['module'] == module)
        chain.append({'receipt': str(path), 'source_sha256': c['source_sha256'],
                      'output_sha256': c['output_sha256'], 'status': c.get('status', 'fresh')})
        if c.get('status') != 'reused_exact_successful_local_output':
            assert c['exit_code'] == 0, (module, path)
            log = path.parent / (module + '.log')
            assert sha(log.read_bytes()) == c['log_sha256']
            records.append(keep(log, Path('evidence/local-runs') / path.parent.name / log.name))
            successes[module] = {'run': path.parent.name, 'command': c,
                                 'receipt_snapshot': receipt_records[str(path)]['snapshot'],
                                 'log_snapshot': str(Path('evidence/local-runs') / path.parent.name / log.name)}
            return chain
        prior = Path(c['prior_receipt'])
        assert sha(prior.read_bytes()) == c['prior_receipt_sha256']
        pc = next(x for x in receipt(prior)['commands'] if x['module'] == module)
        assert pc['source_sha256'] == c['source_sha256']
        assert pc['output_sha256'] == c['output_sha256']
        path = prior

for rel, f in sources.items():
    if not f['solution_closure']:
        continue
    module = rel.removesuffix('.lean').replace('/', '.')
    assert latest[module]['source_sha256'] == f['sha256'], module
    chains[module] = follow(module, run116)

records.extend(receipt_records.values())
(R / 'LOCAL-EVIDENCE.json').write_text(json.dumps({

    'solution_closure_count': len(chains), 'chains': chains, 'successful_origins': successes,
    'bindings': records, 'github_comparator_executed_by_reviewer': False,
}, indent=2) + '\n')

declarations=[]
for rel,f in sorted(sources.items()):
    if not f['solution_closure']:
        continue
    for number,line in enumerate((R/f['snapshot']).read_text().splitlines(),1):
        match=re.match(r'^(private )?(theorem|def|abbrev)\s+(\S+)',line)
        if match:
            declarations.append({'path':rel,'line':number,'private':bool(match[1]),
                                 'kind':match[2],'name':match[3],'source_sha256':f['sha256']})
queries['all_declaration_names']=['rg','-n','\\b(theorem|lemma|def|abbrev)\\s+('+'|'.join(re.escape(x['name']) for x in declarations)+')\\b',str(M/'Mathlib')]

    'declarations':declarations},indent=2)+'\n')
searches=[]
matched=set()
for label, argv in queries.items():
    result=subprocess.run(argv,capture_output=True,text=True,check=False)
    assert result.returncode in (0,1), (label,result.stderr)
    out=R/'evidence/reuse-search'/f'{label}.txt'
    out.parent.mkdir(parents=True,exist_ok=True)
    out.write_text(result.stdout)
    searches.append({'label':label,'argv':argv,'exit_code':result.returncode,
                     'stdout_snapshot':str(out.relative_to(R)),'stdout_sha256':sha(out.read_bytes())})
    for line in result.stdout.splitlines():
        candidate=line.split(':',1)[0]
        if candidate.startswith(str(M)) and candidate.endswith('.lean'):
            matched.add(Path(candidate))
