698: def mulVec [Fintype n] (M : Matrix m n α) (v : n → α) : m → α
699:   | i => (fun j => M i j) ⬝ᵥ v

704: lemma mulVec_apply [Fintype n] (M : Matrix m n α) (v : n → α) (i : m) :
705:     (M *ᵥ v) i = M.row i ⬝ᵥ v := rfl

707: lemma mulVec_apply_eq_sum [Fintype n] (M : Matrix m n α) (v : n → α) (i : m) :
708:     (M *ᵥ v) i = ∑ j : n, M i j * v j := rfl

710: /--
711: `v ᵥ* M` (notation for `vecMul v M`) is the vector-matrix product of vector `v` and matrix `M`,
712: where `v` is seen as a row vector.
713: 
714: The notation has precedence 73, which comes immediately before ` ⬝ᵥ ` for `dotProduct`,
715: so that `v ᵥ* A ⬝ᵥ w ᵥ* B` is parsed as `(v ᵥ* A) ⬝ᵥ (w ᵥ* B)`.
