#!/usr/bin/env python3
"""Read-only source audit and private plan evidence; never invokes Lean or Lake."""
from pathlib import Path
import datetime
import hashlib
import json
import shutil
import subprocess

R = Path(__file__).resolve().parent
P = R.parents[1]
B = P.parents[1]
L = Path('/private/tmp/nla-lean-local-shared-20260916')
M = L / '.lake/packages/mathlib'
W = Path('/private/tmp/nla-lean-next-mi13-worktree')
sha = lambda f: hashlib.sha256(Path(f).read_bytes()).hexdigest()
freeze = B / 'SP15-ROOT-FREEZE-ACCEPTANCE.json'
assert sha(freeze) == '46318471938f7d105a113af3e449856e3a8f52154714dc434f5cb905d76db4ca'
protected = json.loads((P / 'STATEMENT-FREEZE.json').read_text())['frozen_files']
assert len(protected) == 13
for rel, digest in protected.items():
    assert sha(P / rel) == digest, rel
headers = json.loads((P / 'STATEMENT-HEADERS.json').read_text())['contracts'][15:26]
assert len(headers) == 11
for h in headers:
    assert hashlib.sha256(h['header'].encode()).hexdigest() == h['header_sha256']
    assert h['header'] + ' := by' in (P / 'Challenge.lean').read_text()
(R / 'EXACT-HEADERS.json').write_text(json.dumps(headers, indent=2) + '\n')

primary = {

    'Mathlib/Data/Complex/Basic.lean': [(90, 106), (539, 559), (580, 592)],
    'Mathlib/Analysis/Complex/Norm.lean': [(92, 107)],
}
api = {}
for rel, ranges in primary.items():
    f = M / rel
    lines = f.read_text().splitlines()
    excerpt = '\n\n'.join('\n'.join(f'{i}: {lines[i-1]}' for i in range(a, b+1))
                            for a, b in ranges) + '\n'
    dest = R / 'primary' / (rel.replace('/', '_') + '.txt')
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_text(excerpt)
    olean = M / '.lake/build/lib/lean' / Path(rel).with_suffix('.olean')
    assert olean.is_file(), olean
    api[rel] = {'source_sha256': sha(f), 'line_ranges': ranges,
                'excerpt': str(dest.relative_to(R)), 'excerpt_sha256': sha(dest),
                'pinned_cached_olean_present': True}
(R / 'API-BINDINGS.json').write_text(json.dumps(api, indent=2) + '\n')

canonical = {}
commit = 'd348d7471e2ff881ae30fb8a9c40323a61cd383a'
for name in ['README.md', 'solution.md', 'solution.tex']:
    rel = 'eigenvalues-and-inverse-problems/SP-15/' + name
    data = subprocess.check_output(['git', 'show', commit + ':' + rel], cwd=W)
    retained = P / 'sources/canonical' / name
    assert data == retained.read_bytes(), rel
    canonical[rel] = {'commit': commit, 'sha256': sha(retained),
                       'read_full_source': True, 'matches_exact_git_blob': True}

runtime = []
for run, modules in [(79, ['BoxGeometry']), (89, ['CoefficientIdentity']),
                     (91, ['ParameterPositivity', 'SquareRoots']), (102, ['FiniteFiber'])]:
    actual = L / f'runs/development-{run}'
    receipt = json.loads((actual / 'RECEIPT.json').read_text())
    assert receipt['end']
    out = R / f'actual{run}'
    out.mkdir(exist_ok=True)
    shutil.copy2(actual / 'RECEIPT.json', out / 'RECEIPT.json')
    checked = []
    for name in modules:
        module = 'NLA.SP15.' + name
        c = next(c for c in receipt['commands'] if c['module'] == module)
        assert c['exit_code'] == 0
        rel = Path(module.replace('.', '/')).with_suffix('.lean')
        assert sha(P / rel) == c['source_sha256']
        log = actual / (module + '.log')
        assert sha(log) == c['log_sha256']
        text = log.read_text()
        assert 'sorryAx' not in text and 'error:' not in text
        shutil.copy2(log, out / log.name)
        checked.append({'module': module, 'source_sha256': c['source_sha256'],
                        'log_sha256': c['log_sha256'], 'exit_code': 0,
                        'elapsed_seconds': c['elapsed_seconds']})
    runtime.append({'run': run, 'receipt_sha256': sha(actual / 'RECEIPT.json'),
                    'scope': 'Only named successful modules; actual79 also retains an unrelated historical failed Jacobian attempt',
                    'checked': checked})

                       'use': 'Read the actual previously accepted Gram/eigenvalue/singular-value transport; no fresh execution in this plan'}

    'second_pivot': 'D=sI+Q; actual positive definiteness and inverse cancellation proved',

    'new_Lean_execution': False,

(R / 'PLAN-CHECKS.json').write_text(json.dumps({
    'time': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'scope': 'Source-bound pre-code plans for exact frozen16–26 only',
    'freeze_acceptance_sha256': sha(freeze), 'protected_files_checked': protected,
    'exact_headers_checked': len(headers), 'primary_source_files': len(api),

                  'primary_files': len(api), 'canonical_sources_matched': len(canonical),

                  'compiler_executed': False}, indent=2))
