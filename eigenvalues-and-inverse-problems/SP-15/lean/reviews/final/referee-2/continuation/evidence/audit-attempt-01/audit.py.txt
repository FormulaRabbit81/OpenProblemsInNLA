def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def need(condition, label, detail=None):
    checks.append({'check': label, 'pass': bool(condition), 'detail': detail})
    if not condition:
        raise AssertionError((label, detail))

def dump(rel, obj):
    p = HERE / rel
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(obj, indent=2, ensure_ascii=False) + '\n')

def keep(p, rel):
    p, q = Path(p), HERE / rel
    q.parent.mkdir(parents=True, exist_ok=True)
    if q.exists():
        need(sha(q) == sha(p), 'retained-unchanged:' + rel)
    else:
        shutil.copy2(p, q)
    retained[str(p)] = {'original': str(p), 'retained': rel, 'sha256': sha(p)}
    return q

def strip_comments(text):
    """Preserve strings/newlines and token separation; support nested Lean comments."""
    out, i, depth = [], 0, 0
    while i < len(text):
        if depth:
            if text.startswith('/-', i):
                depth += 1; i += 2
            elif text.startswith('-/', i):
                depth -= 1; i += 2; out.append(' ')
            else:
                if text[i] == '\n': out.append('\n')
                i += 1
        elif text.startswith('/-', i):
            out.append(' '); depth = 1; i += 2
        elif text.startswith('--', i):
            j = text.find('\n', i)
            i = len(text) if j < 0 else j
            out.append(' ')
        elif text[i] == '"':
            j = i + 1
            while j < len(text):
                if text[j] == '\\': j += 2
                elif text[j] == '"': j += 1; break
                else: j += 1
            out.append(text[i:j]); i = j
        else:
            out.append(text[i]); i += 1
    assert depth == 0, 'unclosed comment'
    return ''.join(out)

def normalized_code(text):
    return ' '.join(strip_comments(text).split())

def resolve(mod, path, chain=None):
    chain = [] if chain is None else chain
    need(str(path) not in chain, 'acyclic-receipt-chain:' + mod + ':' + str(path))
    r = receipt(path)
    c = next(c for c in r['commands'] if c['module']==mod)
    rel = modules[mod]
    need(c['source_sha256'] == closure[rel], 'receipt-source:' + mod + ':' + Path(path).parent.name)
    need(c['output_sha256'] == root_commands[mod]['output_sha256'], 'receipt-output:' + mod + ':' + Path(path).parent.name)
    transitive_hashes = {modules[dep]: closure[modules[dep]] for dep in transitive(mod)}
    for dep, expected in transitive_hashes.items():
        need(r['source_inputs'][dep] == expected, 'receipt-transitive:' + mod + ':' + dep + ':' + Path(path).parent.name)
    if c.get('status') == 'reused_exact_successful_local_output':
        need(c['transitive_source_hashes'] == transitive_hashes, 'reuse-exact-transitive-map:' + mod)
        prior = Path(c['prior_receipt'])
        need(sha(prior) == c['prior_receipt_sha256'], 'prior-receipt-hash:' + mod + ':' + str(prior))
        return resolve(mod, prior, chain + [str(path)])
    need(c['exit_code'] == 0, 'fresh-success:' + mod)
    need(c['argv'] == [env['compiler'], '--threads=1', '--memory=4096', '-o', str(L/'.lake/build/lib/lean'/(rel[:-5]+'.olean')), rel], 'fresh-exact-command:' + mod)
    need(c['cwd'] == str(L) and r['compiler_sha256'] == root['compiler_sha256'], 'fresh-real-compiler:' + mod)
    need((r['platform'], r['max_compiler_processes'], r['threads'], r['memory_cap_mib']) == ('darwin',1,1,4096), 'fresh-resource-receipt:' + mod)
    log = Path(path).parent/(mod+'.log')
    need(sha(log) == c['log_sha256'], 'fresh-raw-log:' + mod)
    keep(log, 'evidence/runtime/' + log.parent.name + '/' + log.name)
    need(set(c['dependency_olean_sha256']) >= set(imports(mod)), 'fresh-dependency-coverage:' + mod)
    for dep, expected in c['dependency_olean_sha256'].items():
        if dep in modules:
            need(expected == root_commands[dep]['output_sha256'], 'fresh-dependency-output:' + mod + ':' + dep)
    return {'module': mod, 'fresh_run': Path(path).parent.name, 'reuse_chain': chain,
            'fresh_receipt': str(path), 'fresh_receipt_sha256': sha(path), 'fresh_command': c,
            'retained_log': retained[str(log)]['retained']}
