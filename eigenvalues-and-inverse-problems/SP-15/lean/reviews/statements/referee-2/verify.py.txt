#!/usr/bin/env python3
"""Verify sealed private review; --sources also requires unchanged original paths."""
from pathlib import Path
import hashlib,json,sys
R=Path(__file__).resolve().parent
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
m=json.loads((R/'MANIFEST.json').read_text())
for row in m['files']:
    assert sha(R/row['path'])==row['sha256'],row['path']
actual={str(p.relative_to(R)) for p in R.rglob('*') if p.is_file()}
assert actual=={r['path'] for r in m['files']}|{'MANIFEST.json'}
for row in json.loads((R/'RETAINED-COPIES.json').read_text()):
    assert sha(R/row['retained'])==row['sha256'],row['retained']
if '--sources' in sys.argv:
    rows=json.loads((R/'SOURCE-BINDINGS.json').read_text())
    for row in rows:assert sha(row['path'])==row['sha256'],row['path']
    print('PASS',len(m['files']),'sealed payloads and',len(rows),'current original-path bindings')
else:print('PASS',len(m['files']),'sealed payloads; original paths not consulted')
