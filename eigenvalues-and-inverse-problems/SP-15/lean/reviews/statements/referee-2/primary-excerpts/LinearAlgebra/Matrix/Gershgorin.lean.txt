27: /-- **Gershgorin's circle theorem**: for any eigenvalue `μ` of a square matrix `A`, there exists an
28: index `k` such that `μ` lies in the closed ball of center the diagonal term `A k k` and of
29: radius the sum of the norms `∑ j ≠ k, ‖A k j‖`. -/
30: theorem eigenvalue_mem_ball {μ : K} (hμ : Module.End.HasEigenvalue (Matrix.toLin' A) μ) :
31:     ∃ k, μ ∈ Metric.closedBall (A k k) (∑ j ∈ Finset.univ.erase k, ‖A k j‖) := by
32:   cases isEmpty_or_nonempty n
33:   · exfalso
34:     exact hμ Submodule.eq_bot_of_subsingleton
35:   · obtain ⟨v, h_eg, h_nz⟩ := hμ.exists_hasEigenvector
36:     obtain ⟨i, -, h_i⟩ := Finset.exists_mem_eq_sup' Finset.univ_nonempty (fun i => ‖v i‖)
37:     have h_nz : v i ≠ 0 := by
38:       contrapose h_nz
39:       ext j
40:       rw [Pi.zero_apply, ← norm_le_zero_iff]
41:       refine (h_i ▸ Finset.le_sup' (fun i => ‖v i‖) (Finset.mem_univ j)).trans ?_
42:       exact norm_le_zero_iff.mpr h_nz
43:     have h_le : ∀ j, ‖v j * (v i)⁻¹‖ ≤ 1 := fun j => by
44:       rw [norm_mul, norm_inv, mul_inv_le_iff₀ (norm_pos_iff.mpr h_nz), one_mul]
45:       exact h_i ▸ Finset.le_sup' (fun i => ‖v i‖) (Finset.mem_univ j)
46:     simp_rw [mem_closedBall_iff_norm']
47:     refine ⟨i, ?_⟩
48:     calc
