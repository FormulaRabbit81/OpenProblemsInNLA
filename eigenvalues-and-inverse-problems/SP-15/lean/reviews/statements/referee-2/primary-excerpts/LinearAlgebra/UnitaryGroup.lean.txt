57: /-- `Matrix.unitaryGroup n` is the group of `n` by `n` matrices where the star-transpose is the
58: inverse.
59: -/
60: abbrev unitaryGroup : Submonoid (Matrix n n α) :=
61:   unitary (Matrix n n α)
62:

65: example : StarMul (unitaryGroup n α) := inferInstance
66: 
67: end

72: theorem mem_unitaryGroup_iff : A ∈ Matrix.unitaryGroup n α ↔ A * star A = 1 := by
73:   refine ⟨And.right, fun hA => ⟨?_, hA⟩⟩
74:   simpa only [mul_eq_one_comm] using hA

76: theorem mem_unitaryGroup_iff' : A ∈ Matrix.unitaryGroup n α ↔ star A * A = 1 := by

78:   rwa [mul_eq_one_comm] at hA

80: theorem det_of_mem_unitary {A : Matrix n n α} (hA : A ∈ Matrix.unitaryGroup n α) :
81:     A.det ∈ unitary α := by

83:   · simpa [star, det_transpose] using congr_arg det hA.1
