83: /--
84: If `T : E →ₗ[𝕜] F` is a linear map between finite dimensional inner product spaces, then
85: `T.singularValues` is the infinite sequence where the first dim(E) elements are the square roots of
86: eigenvalues of `T.adjoint ∘ₗ T` (which are guaranteed to be nonnegative real numbers), arranged
87: in descending order and repeated according to their multiplicity, and the rest of the elements in
88: the infinite sequence are zero. Please see the module docstring of
89: `Mathlib/Analysis/InnerProductSpace/SingularValues.lean` for an explanation of this design decision.
90: 
91: The singular values are zero-indexed, so `T.singularValues 0` refers to the first singular value.
92: This means the positive singular values occur at `0 ≤ i < rank(T)` and not `1 ≤ i ≤ rank(T)`.
93: -/
94: noncomputable def singularValues : ℕ →₀ ℝ :=
95:   Finsupp.embDomain Fin.valEmbedding <|
96:     Finsupp.ofSupportFinite
97:       (fun i ↦ √(T.isSymmetric_adjoint_comp_self.eigenvalues rfl i))
98:       (Set.toFinite _)
99: 
100: theorem singularValues_nonneg (i : ℕ) : 0 ≤ T.singularValues i := by
101:   rw [singularValues, Finsupp.embDomain_apply, Finsupp.ofSupportFinite_coe]
102:   split_ifs <;> positivity
103: 
104: theorem singularValues_pos_iff_ne_zero (i : ℕ) :
105:     0 < T.singularValues i ↔ T.singularValues i ≠ 0 := by
106:   grind [T.singularValues_nonneg i]

108: /--
109: Connection between `LinearMap.singularValues` and `LinearMap.IsSymmetric.eigenvalues`.
110: Together with `LinearMap.singularValues_of_finrank_le`, this characterizes the singular values.
111: 
112: Because of the square root, you probably need to use
113: `T.isPositive_adjoint_comp_self.nonneg_eigenvalues` to make effective use of this theorem.
114: -/
115: theorem singularValues_fin {n : ℕ} (hn : finrank 𝕜 E = n) (i : Fin n) :
116:     T.singularValues i = √(T.isSymmetric_adjoint_comp_self.eigenvalues hn i) := by
117:   subst hn
118:   exact Finsupp.embDomain_apply_self _ _ i

120: theorem singularValues_of_lt {n : ℕ} (hn : finrank 𝕜 E = n) {i : ℕ} (hi : i < n) :
121:     T.singularValues i = √(T.isSymmetric_adjoint_comp_self.eigenvalues hn ⟨i, hi⟩) :=
