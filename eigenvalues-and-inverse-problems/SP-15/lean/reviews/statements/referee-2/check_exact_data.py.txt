class Poly:
    def __init__(self,d=0):
        self.d={k:F(v) for k,v in d.items() if v} if isinstance(d,dict) else ({ZERO:F(d)} if d else {})
    def __add__(self,b):
        b=cast(b);d=self.d.copy()
        for k,v in b.d.items():d[k]=d.get(k,F(0))+v
        return Poly(d)
    __radd__=__add__
    def __neg__(self):return Poly({k:-v for k,v in self.d.items()})
    def __sub__(self,b):return self+-cast(b)
    def __rsub__(self,b):return cast(b)+-self
    def __mul__(self,b):
        b=cast(b);d={}
        for a,av in self.d.items():
            for be,bv in b.d.items():
                e=tuple(x+y for x,y in zip(a,be));sign=(-1)**(e[-1]//2);e=e[:-1]+(e[-1]%2,)

        return Poly(d)
    __rmul__=__mul__
    def __pow__(self,k):
        assert isinstance(k,int) and k>=0
        r=Poly(1)
        for _ in range(k):r=r*self
        return r
    def partial(self,i):
        d={}
        for e,v in self.d.items():
            if e[i]:

        return Poly(d)
    def at(self,x):
        assert len(x)==10 and all(not any(e[10:]) for e in self.d)
        return sum((v*product(F(xi)**power for xi,power in zip(x,e)) for e,v in self.d.items()),F(0))
    def __eq__(self,b):return self.d==cast(b).d

def translate(t):

def expression(t,env):
    def run(n):
        if isinstance(n,ast.Expression):return run(n.body)
        if isinstance(n,ast.Constant) and isinstance(n.value,int):return F(n.value)
        if isinstance(n,ast.Name):return env[n.id]
        if isinstance(n,ast.Subscript) and isinstance(n.value,ast.Name) and n.value.id=='x':return env['x'][int(run(n.slice))]
        if isinstance(n,ast.UnaryOp) and isinstance(n.op,ast.USub):return -run(n.operand)
        if isinstance(n,ast.BinOp):
            a=run(n.left);b=run(n.right)
            if isinstance(n.op,ast.Add):return a+b
            if isinstance(n.op,ast.Sub):return a-b
            if isinstance(n.op,ast.Mult):return a*b
            if isinstance(n.op,ast.Div):return a/b
            if isinstance(n.op,ast.Pow):return a**int(b)
        raise ValueError(ast.dump(n))
    return run(ast.parse(translate(' '.join(t.split())),mode='eval'))
