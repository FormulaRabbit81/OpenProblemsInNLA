def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def check(label, truth):
    if not truth: raise AssertionError(label)
    checks.append(label)
def bind(p, expected=None, why='Byte authentication; not a full-text read claim'):
    p=Path(p).absolute(); raw=p.read_bytes(); digest=hashlib.sha256(raw).hexdigest()
    if expected is not None: check('SHA256 '+str(p),digest==expected)
    key=str(p)
    if key in bindings: check('Stable re-read '+key,bindings[key]['sha256']==digest)
    else: bindings[key]={'path':key,'sha256':digest,'bytes':len(raw),'scope':why}
    return digest
def read(p):
    bind(p); return json.loads(Path(p).read_text())
def retain(p, rel):
    p=Path(p); digest=bind(p); out=R/'evidence'/rel
    check('Retained path remains in review '+str(rel),out.resolve().is_relative_to(R))
    out.parent.mkdir(parents=True,exist_ok=True); out.write_bytes(p.read_bytes())
    check('Exact retained bytes '+str(rel),sha(out)==digest)
    copies.append({'source':str(p.absolute()),'retained':str(out.relative_to(R)),
                   'sha256':digest,'bytes':p.stat().st_size})
def manifest(base, name='MANIFEST.json', expected=None):
    mp=base/name; bind(mp,expected); data=read(mp); rows=data['files']
    entries=({r['path']:r['sha256'] for r in rows} if isinstance(rows,list)
             else {k:(v['sha256'] if isinstance(v,dict) else v) for k,v in rows.items()})
    for name,digest in entries.items():
        check('Safe manifest path '+name,not Path(name).is_absolute() and '..' not in Path(name).parts)
        bind(base/name,digest)
    return data,entries
