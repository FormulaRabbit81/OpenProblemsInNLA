368: /-- Determinant of a 2×2 block matrix, expanded around an invertible top left element in terms of
369: the Schur complement. -/
370: theorem det_fromBlocks₁₁ (A : Matrix m m α) (B : Matrix m n α) (C : Matrix n m α)
371:     (D : Matrix n n α) [Invertible A] :
372:     (Matrix.fromBlocks A B C D).det = det A * det (D - C * ⅟A * B) := by
373:   rw [fromBlocks_eq_of_invertible₁₁ (A := A), det_mul, det_mul, det_fromBlocks_zero₂₁,
374:     det_fromBlocks_zero₂₁, det_fromBlocks_zero₁₂, det_one, det_one, one_mul, one_mul, mul_one]
375: 
376: @[simp]
377: theorem det_fromBlocks_one₁₁ (B : Matrix m n α) (C : Matrix n m α) (D : Matrix n n α) :
378:     (Matrix.fromBlocks 1 B C D).det = det (D - C * B) := by
379:   have : Invertible (1 : Matrix m m α) := invertibleOne
380:   rw [det_fromBlocks₁₁, invOf_one, Matrix.mul_one, det_one, one_mul]

382: /-- Determinant of a 2×2 block matrix, expanded around an invertible bottom right element in terms
383: of the Schur complement. -/
384: theorem det_fromBlocks₂₂ (A : Matrix m m α) (B : Matrix m n α) (C : Matrix n m α)
385:     (D : Matrix n n α) [Invertible D] :
386:     (Matrix.fromBlocks A B C D).det = det D * det (A - B * ⅟D * C) := by
387:   have : fromBlocks A B C D =
388:       (fromBlocks D C B A).submatrix (Equiv.sumComm _ _) (Equiv.sumComm _ _) := by
389:     ext (i j)
390:     cases i <;> cases j <;> rfl
391:   rw [this, det_submatrix_equiv_self, det_fromBlocks₁₁]

393: @[simp]
394: theorem det_fromBlocks_one₂₂ (A : Matrix m m α) (B : Matrix m n α) (C : Matrix n m α) :
395:     (Matrix.fromBlocks A B C 1).det = det (A - B * C) := by
396:   have : Invertible (1 : Matrix n n α) := invertibleOne
397:   rw [det_fromBlocks₂₂, invOf_one, Matrix.mul_one, det_one, one_mul]

399: /-- The **Weinstein–Aronszajn identity**. Note the `1` on the LHS is of shape m×m, while the `1` on
400: the RHS is of shape n×n. -/
401: theorem det_one_add_mul_comm (A : Matrix m n α) (B : Matrix n m α) :
402:     det (1 + A * B) = det (1 + B * A) :=
403:   calc
404:     det (1 + A * B) = det (fromBlocks 1 (-A) B 1) := by
405:       rw [det_fromBlocks_one₂₂, Matrix.neg_mul, sub_neg_eq_add]
406:     _ = det (1 + B * A) := by rw [det_fromBlocks_one₁₁, Matrix.mul_neg, sub_neg_eq_add]
407: 
408: /-- Alternate statement of the **Weinstein–Aronszajn identity** -/
409: theorem det_mul_add_one_comm (A : Matrix m n α) (B : Matrix n m α) :
410:     det (A * B + 1) = det (B * A + 1) := by rw [add_comm, det_one_add_mul_comm, add_comm]
411: 
412: theorem det_one_sub_mul_comm (A : Matrix m n α) (B : Matrix n m α) :
413:     det (1 - A * B) = det (1 - B * A) := by
414:   rw [sub_eq_add_neg, ← Matrix.neg_mul, det_one_add_mul_comm, Matrix.mul_neg, ← sub_eq_add_neg]
