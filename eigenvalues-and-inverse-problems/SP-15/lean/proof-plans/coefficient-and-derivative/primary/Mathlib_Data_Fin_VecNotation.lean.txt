185: open Qq in
186: /-- A simproc that handles terms of the form `Matrix.vecCons a f i` where `i` is a numeric literal.
187: 
188: In practice, this is most effective at handling `![a, b, c] i`-style terms. -/
189: dsimproc cons_val (Matrix.vecCons _ _ _) := fun e => do
190:   let_expr Matrix.vecCons α en x xs' ei := ← Meta.whnfR e | return .continue
191:   let some i := ei.int? | return .continue
192:   let (xs, etailn, tail) ← matchVecConsPrefix en xs'
193:   let xs := x :: xs
194:   -- Determine if the tail is a numeral or only an offset.
195:   let (tailn, variadic, etailn) ← do
196:     let etailn_whnf : Q(ℕ) ← Meta.whnfD etailn
197:     if let Expr.lit (.natVal length) := etailn_whnf then
198:       pure (length, false, q(OfNat.ofNat $etailn_whnf))
199:     else if let some ((base : Q(ℕ)), offset) ← (Meta.isOffset? etailn_whnf).run then
200:       pure (offset, true, q($base + $offset))
201:     else
202:       pure (0, true, etailn)
203:   -- Wrap the index if possible, and abort if not
204:   let wrapped_i ←
205:     if variadic then
206:       -- can't wrap as we don't know the length
207:       unless 0 ≤ i ∧ i < xs.length + tailn do return .continue
208:       pure i.toNat
209:     else
210:       pure (i % (xs.length + tailn)).toNat
211:   if h : wrapped_i < xs.length then
212:     return .continue xs[wrapped_i]
213:   else
214:     -- Within the `tail`
215:     let _ ← synthInstanceQ q(NeZero $etailn)
216:     have i_lit : Q(ℕ) := mkRawNatLit (wrapped_i - xs.length)
217:     return .continue (.some <| .app tail q(OfNat.ofNat $i_lit : Fin $etailn))
218: 
219: end simprocs
