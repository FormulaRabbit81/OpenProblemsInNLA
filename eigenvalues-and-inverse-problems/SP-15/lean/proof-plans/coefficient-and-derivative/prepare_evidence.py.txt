class Dual:
    """Exact integer value and the ten formal first-derivative coefficients."""
    def __init__(self, value, gradient=None):
        self.value = value
        self.gradient = tuple(gradient or [0] * 10)
    @staticmethod
    def wrap(a): return a if isinstance(a, Dual) else Dual(a)
    def __add__(self, b):
        b = self.wrap(b)
        return Dual(self.value + b.value, [x+y for x,y in zip(self.gradient,b.gradient)])
    __radd__ = __add__
    def __neg__(self): return Dual(-self.value, [-x for x in self.gradient])
    def __sub__(self, b): return self + -self.wrap(b)
    def __mul__(self, b):
        b = self.wrap(b)
        return Dual(self.value*b.value,
                    [self.value*y+b.value*x for x,y in zip(self.gradient,b.gradient)])
    __rmul__ = __mul__
    def __pow__(self, n):
        assert isinstance(n, int) and n >= 0
        out = Dual(1)
        for _ in range(n): out = out * self
        return out

env = {'x': xs}
translate = str.maketrans({'₁':'1','₂':'2','₃':'3'})
def parse_expr(s):
    s = s.translate(translate).replace('^', '**')
    s = re.sub(r'\bx\s+(\d+)\b', r'x[\1]', s)
    return ast.parse(' '.join(s.split()), mode='eval').body
def ev(node):
    if isinstance(node, ast.Constant):
        assert isinstance(node.value, int)
        return node.value
    if isinstance(node, ast.Name): return env[node.id]
    if isinstance(node, ast.Subscript):
        assert isinstance(node.value, ast.Name) and node.value.id == 'x'
        return xs[ev(node.slice)]
    if isinstance(node, ast.UnaryOp) and isinstance(node.op, ast.USub): return -ev(node.operand)
    if isinstance(node, ast.BinOp):
        a,b = ev(node.left), ev(node.right)
        if isinstance(node.op, ast.Add): return a+b
        if isinstance(node.op, ast.Sub): return a-b
        if isinstance(node.op, ast.Mult): return a*b
        if isinstance(node.op, ast.Pow): return a**b
    raise AssertionError(ast.dump(node))
