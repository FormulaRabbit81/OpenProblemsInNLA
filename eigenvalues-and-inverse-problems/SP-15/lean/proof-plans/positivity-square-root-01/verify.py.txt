#!/usr/bin/env python3
"""Verify sealed review; --sources requires current original bound paths."""
from pathlib import Path
import hashlib,json,sys
R=Path(__file__).resolve().parent
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
m=json.loads((R/'MANIFEST.json').read_text())
for name,h in m['files'].items():assert sha(R/name)==h,name
assert {str(p.relative_to(R)) for p in R.rglob('*') if p.is_file()}==set(m['files'])|{'MANIFEST.json'}
for row in json.loads((R/'COPY-MAP.json').read_text()):assert sha(R/row['retained'])==row['sha256'],row['retained']
if '--sources' in sys.argv:
    bindings=json.loads((R/'BINDINGS.json').read_text())
    for p,h in bindings.items():assert sha(p)==h,p
    print('PASS',len(bindings),'current source/evidence bindings')
print('PASS',len(m['files']),'sealed review payloads')
