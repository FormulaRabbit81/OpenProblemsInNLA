42: /-- A matrix is Hermitian if it is equal to its conjugate transpose. On the reals, this definition
43: captures symmetric matrices. -/
44: def IsHermitian (A : Matrix n n α) : Prop := Aᴴ = A
45: 
46: instance (A : Matrix n n α) [Decidable (Aᴴ = A)] : Decidable (IsHermitian A) :=
47:   inferInstanceAs <| Decidable (_ = _)
48: 
49: theorem IsHermitian.eq {A : Matrix n n α} (h : A.IsHermitian) : Aᴴ = A := h
50: 
51: theorem isHermitian_iff_isSelfAdjoint {A : Matrix n n α} :
52:     A.IsHermitian ↔ IsSelfAdjoint A := Iff.rfl
53: 
54: protected alias ⟨IsHermitian.isSelfAdjoint, _root_.IsSelfAdjoint.isHermitian⟩ :=
55:   isHermitian_iff_isSelfAdjoint
56: 
57: theorem IsHermitian.star_eq (hA : A.IsHermitian) : star A = A := hA.isSelfAdjoint.star_eq
58: 
59: theorem IsHermitian.ext {A : Matrix n n α} : (∀ i j, star (A j i) = A i j) → A.IsHermitian := by
60:   intro h; ext i j; exact h i j
61: 
62: theorem IsHermitian.apply {A : Matrix n n α} (h : A.IsHermitian) (i j : n) : star (A j i) = A i j :=
63:   congr_fun (congr_fun h _) _
64: 
65: theorem IsHermitian.ext_iff {A : Matrix n n α} : A.IsHermitian ↔ ∀ i j, star (A j i) = A i j :=
66:   ⟨IsHermitian.apply, IsHermitian.ext⟩
