59: /-- Let `b` map rows and columns of a square matrix `M` to blocks indexed by `α`s. Then
60: `BlockTriangular M n b` says the matrix is block triangular. -/
61: def BlockTriangular (M : Matrix m m R) (b : m → α) : Prop :=
62:   ∀ ⦃i j⦄, b j < b i → M i j = 0
