import NLA.SP15.Numerical
import Mathlib.LinearAlgebra.Matrix.Block
import Mathlib.Tactic.FinCases

set_option autoImplicit false
set_option leancert.trust "kernel"

namespace NLA.SP15
noncomputable section
open scoped BigOperators Matrix

theorem coefficient_base_value : coefficients basePoint = coefficientBaseValue := by
  funext i
  fin_cases i <;> norm_num [coefficients, basePoint, coefficientBaseValue]

private theorem jacobian_lu_reconstruction :
    jacobianMinor = lowerCertificate * upperCertificate := by
  ext i j
  -- Literal indices and triangular zeros simplify before rational arithmetic.
  fin_cases i <;> fin_cases j <;>
    norm_num [jacobianMinor, jacobian, lowerCertificate, upperCertificate,
      Matrix.mul_apply, Fin.sum_univ_succ]

theorem jacobian_lu_certificate :
    jacobianMinor = lowerCertificate * upperCertificate ∧
    (∀ i j : Fin 9, i < j → lowerCertificate i j = 0) ∧
    (∀ i : Fin 9, lowerCertificate i i = 1) ∧
    (∀ i j : Fin 9, j < i → upperCertificate i j = 0) ∧
    (∀ i : Fin 9, upperCertificate i i ≠ 0) ∧
    (∏ i : Fin 9, upperCertificate i i) = (-1088 : ℝ) := by
  refine ⟨jacobian_lu_reconstruction, ?_, ?_, ?_, ?_, ?_⟩
  · intro i j hij
    fin_cases i <;> fin_cases j <;> norm_num [lowerCertificate] at hij ⊢
  · intro i
    fin_cases i <;> norm_num [lowerCertificate]
  · intro i j hij
    fin_cases i <;> fin_cases j <;> norm_num [upperCertificate] at hij ⊢
  · intro i
    fin_cases i <;> norm_num [upperCertificate]
  · norm_num [upperCertificate, Fin.prod_univ_succ]

theorem jacobian_minor_nonsingular : jacobianMinor.det = (-1088 : ℝ) := by
  obtain ⟨hreconstruct, hLower, hLowerDiag, hUpper, _, hProduct⟩ := jacobian_lu_certificate
  have hL : lowerCertificate.IsLowerTriangular := by
    intro i j hij
    exact hLower i j hij
  have hU : upperCertificate.IsUpperTriangular := by
    intro i j hij
    exact hUpper i j hij
  rw [hreconstruct, Matrix.det_mul, Matrix.det_of_isLowerTriangular lowerCertificate hL,
    Matrix.det_of_isUpperTriangular hU, hProduct]
  simp [hLowerDiag]

#print axioms coefficient_base_value
#assert_trust kernel coefficient_base_value
#print axioms jacobian_lu_certificate
#assert_trust kernel jacobian_lu_certificate
#print axioms jacobian_minor_nonsingular
#assert_trust kernel jacobian_minor_nonsingular

end
end NLA.SP15
