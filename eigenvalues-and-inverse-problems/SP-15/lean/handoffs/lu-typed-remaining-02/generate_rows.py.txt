#!/usr/bin/env python3
"""Emit typed scalar goals; Lean must prove every link to the frozen tables."""
from pathlib import Path
from fractions import Fraction
import datetime, hashlib, json, re, sys

R = Path(__file__).resolve().parent
P = R.parents[1]
source = P / 'NLA/SP15/Definitions.lean'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert sha(source) == 'fa57c71cd3e6d6551f081e479121f9f6a4695858732f3e6d1fa4cd79ad927498'
s = source.read_text()
def table(name):
    raw = re.search(r'def '+name+r' :.*?:=\s*!!\[(.*?)\]\n', s, re.S).group(1)
    return [[t.strip() for t in row.split(',')] for row in raw.split(';')]
J,L,U = table('jacobian'),table('lowerCertificate'),table('upperCertificate')
assert len(J)==len(L)==len(U)==9
assert all(len(row)==10 for row in J)
assert all(len(row)==9 for row in L+U)
rows = [int(a) for a in sys.argv[1:]]
assert rows and all(0<=i<9 for i in rows) and len(set(rows))==len(rows)
out = {}
for i in rows:
    pieces = [f'''/-
Copyright (c) 2026 George Stepaniants. Released under Apache 2.0 license.
Department of Computing and Mathematical Sciences, California Institute of Technology.
Substantial OpenAI Codex assistance. Prior mathematical attribution is retained.

Row {i} of the frozen LU identity. Every typed change checks all nine actual
table entries by definitional equality before sparse rational arithmetic.
No untrusted numerical output is imported as a premise.
-/
import NLA.SP15.Numerical
import Mathlib.Tactic.FinCases

set_option autoImplicit false
set_option leancert.trust "kernel"

namespace NLA.SP15
noncomputable section
open scoped BigOperators Matrix

theorem jacobian_lu_row_{i} (j : Fin 9) :
    jacobianMinor {i} j = (lowerCertificate * upperCertificate) {i} j := by
  rw [Matrix.mul_apply]
  simp only [Fin.sum_univ_succ, Fin.sum_univ_zero]
  fin_cases j
''']
    for j in range(9):
        assert sum(Fraction(L[i][k])*Fraction(U[k][j]) for k in range(9)) == Fraction(J[i][j])
        rhs='0'
        for k in reversed(range(9)):
            a=L[i][k].replace('/',' / ');b=U[k][j].replace('/',' / ')
            rhs=f'(({a}) * ({b}) + {rhs})'
        pieces.append(f'  · -- Column {j}: retain every product; norm_num then removes zero terms.\n')
        pieces.append(f'    change ({J[i][j]} : ℝ) =\n      {rhs}\n    norm_num\n')
    pieces.append(f'\n#print axioms jacobian_lu_row_{i}\n#assert_trust kernel jacobian_lu_row_{i}\n\nend\nend NLA.SP15\n')
    dest=P/f'NLA/SP15/JacobianRows/Row{i}.lean'
    dest.parent.mkdir(parents=True,exist_ok=True)
    assert not dest.exists(),dest
    dest.write_text(''.join(pieces))
    out[str(dest.relative_to(P))]={'sha256':sha(dest),'row':i,'columns':list(range(9)),
                                 'literal_products_per_column':9,
                                 'nonzero_products_total':sum(Fraction(L[i][k])!=0 and Fraction(U[k][j])!=0 for j in range(9) for k in range(9))}
receipt={'time':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'scope':'Source-derived candidate emission only; actual Lean execution pending',
 'definitions_sha256':sha(source),'generator_sha256':sha(Path(__file__)),
 'rows':out,'all_numeric_diagnostics_exact':True,'compiler_executed':False,'count_change':0}
receipt_path=R/('GENERATION-'+','.join(map(str,rows))+'.json')
assert not receipt_path.exists(),receipt_path
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps(receipt,indent=2))
