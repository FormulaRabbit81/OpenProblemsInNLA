def variable(i):

def add(a,b):
    c=dict(a)
    for e,v in b.items():
        c[e]=c.get(e,0)+v
        if c[e]==0: del c[e]
    return c
def neg(a): return {e:-v for e,v in a.items()}

    c={}
    for e,v in a.items():
        for f,w in b.items():
            g=[x+y for x,y in zip(e,f)]
            sign=(-1)**(g[-1]//2);g[-1]%=2;g=tuple(g)
            c[g]=c.get(g,0)+sign*v*w
            if c[g]==0:del c[g]
    return c

def rref(a):

def determinant(a):
    a=[[Fraction(x) for x in row] for row in a];det=Fraction(1)
    for c in range(len(a)):

        if k is None:return Fraction(0)
        if k!=c:a[k],a[c]=a[c],a[k];det=-det
        t=a[c][c];det*=t
        for i in range(c+1,len(a)):
            ratio=a[i][c]/t
            a[i]=[x-ratio*y for x,y in zip(a[i],a[c])]
    return det
def matrix_mul(a,b):
