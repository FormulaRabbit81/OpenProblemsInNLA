B = Path('/private/tmp/nla-lean-next-20260915')

P = B / 'NM04-canonical-package-local187'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
now = datetime.datetime.now(datetime.timezone.utc).isoformat()
assert not P.exists()
assert sha(B / 'NM04-LOCAL-COMPLETE-187.json') == 'd32daefe49cde06358205c59a897a439f3af49aa8d017b31a68f8265893efa4d'

    assert sha(OLD / rel) == h, rel

history = P / 'verification/packaging/local184'
history.mkdir(parents=True)
for rel in ['PACKAGE-MANIFEST.json', 'verification/RETAINED-PATHS.json',
            'README.md', 'formalization.yaml', 'STATE.json',
            'ACTIVE-SOURCE-MANIFEST.json', 'IMPLEMENTATION-MAP.json']:
    dest = history / rel
    dest.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(OLD / rel, dest)
records = {r['destination']: r for r in json.loads((P / 'verification/RETAINED-PATHS.json').read_text())}

def copy(src, rel):
    src = Path(src)
    dest = P / rel
    dest.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(src, dest)
    assert sha(dest) == sha(src)
    records[rel] = {'source': str(src), 'destination': rel, 'sha256': sha(src)}

local = json.loads((B / 'NM04-LOCAL-COMPLETE-187.json').read_text())
for rel, h in local['sources'].items():
    src = B / 'NM04-local187-source-snapshot' / rel
    assert sha(src) == h == sha(N / rel)
    copy(src, rel)
copy(B / 'NM04-LOCAL-COMPLETE-187.json', 'verification/LOCAL-COMPLETE.json')
copy(B / 'NM04-LOCAL-COMPLETE-187.json', 'verification/LOCAL-COMPLETE-187.json')

copy(N / 'statement-draft-01/MANIFEST.json', 'verification/statement-history/STATEMENT-DRAFT-MANIFEST.json')

def archive(src, dest):
    with dest.open('wb') as raw, gzip.GzipFile(fileobj=raw, mode='wb', mtime=0) as gz:
        with tarfile.open(fileobj=gz, mode='w') as tar:
            for f in sorted(src.rglob('*')):
                assert not f.is_symlink(), f
                if f.is_file():
                    data = f.read_bytes()
                    text = data.decode('utf-8', errors='ignore')
                    assert not re.search(r'(?i)(?:george|g?stepaniants|sgstepaniants)[^\s@<>]{0,40}@[\w.-]+\.[A-Za-z]{2,}', text), f
                    info = tarfile.TarInfo(src.name + '/' + str(f.relative_to(src)))
                    info.size, info.mode = len(data), 0o644
                    tar.addfile(info, io.BytesIO(data))

archive(N / 'statement-draft-01', P / 'verification/statement-history/statement-draft-01.tar.gz')

R.mkdir(exist_ok=False)

for label, name in [('statement-1', 'NM04-root-statement-20260917'),
                    ('statement-2', 'NM04-statement-formal-review-20260917'),

    manifest = json.loads((src / 'MANIFEST.json').read_text())['files']

        h = v['sha256'] if isinstance(v, dict) else v
        assert sha(src / rel) == h, (name, rel)
    packet = R / (label + '-packet.tar.gz')

                  'verdict': label + '-verdict.json', 'manifest': label + '-manifest.json',
                  'manifest_sha256': sha(src / 'MANIFEST.json'), 'archive': packet.name,
                  'archive_sha256': sha(packet), 'archive_bytes': packet.stat().st_size})
