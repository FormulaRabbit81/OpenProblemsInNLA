"""Publish exact locally checked/reviewed MF18 sources for final Linux verification."""
from pathlib import Path
import datetime, hashlib, importlib.util, json, os, shutil, subprocess

B = Path('/private/tmp/nla-lean-next-20260915')
W = Path('/private/tmp/nla-lean-next-mf18-worktree')
P = B / 'MF18-canonical-package-local340'
REL = 'matrix-functions-and-stability/MF-18/lean'
D = W / REL
SEED = '6ae07217f790f62c7d0550f5d4b30a67500a94c2'
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
load = lambda p: json.loads(Path(p).read_text())
git = lambda *a: subprocess.check_output(['git', *a], cwd=W)
assert git('rev-parse', 'HEAD').decode().strip() == SEED
assert not git('status', '--porcelain').strip() and not D.exists()
preflight = load(B / 'MF18-ROOT-PACKAGE-PREFLIGHT.json')
assert preflight['package_manifest_sha256'] == sha(P / 'PACKAGE-MANIFEST.json') == '1cc99e76f9b4b0abafe645bcd2e29ec21fe3b590e2c9b04635bb67068731c11f'
assert preflight['checks'] == 2035 and preflight['schema']['exit_code'] == 0
original = load(P / 'PACKAGE-MANIFEST.json')['files']
for rel, h in original.items():
    assert sha(P / rel) == h, rel
assert len(load(P / 'reviews/INDEX.json')) == 6
shutil.copytree(P, D)
for name in ['MF18-ROOT-PACKAGE-PREFLIGHT.json']:
    shutil.copy2(B / name, D / 'verification' / name)
shutil.copy2(B / 'preflight_mf18_package.py', D / 'verification/packaging/preflight_mf18_package.py.txt')
shutil.copy2(Path(__file__), D / 'verification/packaging/push_mf18_reviewed_source.py.txt')
shutil.copy2(D / 'PACKAGE-MANIFEST.json', D / 'verification/packaging/reviewed340-PACKAGE-MANIFEST.json')
meta = load(D / 'formalization.yaml')
assert meta['status']['whole_problem_verified'] is False
meta['verification']['package_preflight'] = {'result': 'Coordinator preflight passed',
    'record': 'verification/MF18-ROOT-PACKAGE-PREFLIGHT.json', 'checks': 2035,
    'scope': 'Author/coordinator packaging checks, not an additional independent referee. Two independent complete source approvals remain in reviews/INDEX.json.'}
(D / 'formalization.yaml').write_text(json.dumps(meta, indent=2) + '\n')
readme = (D / 'README.md').read_text()
readme += '\nThe [coordinator publication preflight](verification/MF18-ROOT-PACKAGE-PREFLIGHT.json) checks the complete package, current upstream source correspondence, dependency pins and privacy. It is separate from the two independent full mathematical/source reviews and does not claim Linux execution.\n'
(D / 'README.md').write_text(readme)
for rel, h in original.items():
    if rel not in {'README.md', 'formalization.yaml'}:
        assert sha(D / rel) == h, rel
for rel, h in load(D / 'verification/local340/LOCAL-COMPLETE.json')['sources'].items():
    assert sha(D / rel) == h
manifest = {'stage': 'Complete local340 proof with two independent full-source approvals; actual Linux verification pending',
    'reviewed_package_manifest_sha256': preflight['package_manifest_sha256'],
    'files': {str(f.relative_to(D)): sha(f) for f in sorted(D.rglob('*')) if f.is_file() and f != D / 'PACKAGE-MANIFEST.json'}}
(D / 'PACKAGE-MANIFEST.json').write_text(json.dumps(manifest, indent=2) + '\n')
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1', PYTHONPATH='/tmp/nla-publication-python-20260917',
           GIT_AUTHOR_NAME='George Stepaniants', GIT_AUTHOR_EMAIL='',
           GIT_COMMITTER_NAME='George Stepaniants', GIT_COMMITTER_EMAIL='')
record = {'time': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'seed': SEED,
    'base': 'e7519c46fd249a6a033bfe5d11c66bf47f7f8885', 'project': REL,
    'source_matches_local340': True, 'two_independent_full_source_approvals': True,
    'GitHub_Comparator_run': False, 'canonical_status': 'Solved', 'count_change': 0, 'commands': []}
out = B / 'MF18-PROOF-PUSH-20260918.json'
assert not out.exists()
def run(argv):
    r = subprocess.run(argv, cwd=W, env=env, capture_output=True, text=True)
    record['commands'].append({'argv': argv, 'exit_code': r.returncode, 'stdout': r.stdout, 'stderr': r.stderr})
    out.write_text(json.dumps(record, indent=2) + '\n')
    assert r.returncode == 0, (argv, r.stdout, r.stderr)
    return r.stdout
run(['/Users/georgestepaniants/miniforge3/bin/python', 'tools/lean/validate_manifest.py', REL])
run(['git', 'add', '--force', '--', REL])
run(['git', '-c', 'commit.gpgsign=false', 'commit', '-m', 'Formalize the complete MF-18 complex Green-function rank equality in Lean'])
commit = git('rev-parse', 'HEAD').decode().strip()
assert git('show', '-s', '--format=%ae|%ce', 'HEAD') == b'|\n'
changed = git('diff', '--name-only', '--no-renames', SEED, commit).decode().splitlines()
assert set(changed) == {REL + '/' + str(f.relative_to(D)) for f in D.rglob('*') if f.is_file()}
spec = importlib.util.spec_from_file_location('mf18_selection', W / 'tools/lean/projects.py')
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
projects = module.discover_at_ref(W, commit)
module.require_retained_projects(projects, module.discover_at_ref(W, SEED))
selected = module.select(projects, changed)
assert selected == [{'id': 'MF-18', 'project': REL}], selected
assert git('diff', 'ff6abf718126ceb23f933cf4f627f95104461fe8', commit, '--', 'tools/lean', '.github/workflows/lean-verification.yml') == b''
record.update(proof_commit=commit, tracked_project_files=len(changed), selection=selected,
              selection_scope='Actual committed Git trees and unchanged selector', tools_changed=False,
              package_manifest_sha256=sha(D / 'PACKAGE-MANIFEST.json'))
out.write_text(json.dumps(record, indent=2) + '\n')
run(['git', 'push', 'origin', 'HEAD:refs/heads/codex/lean-mf18-verification'])
record['pushed'] = True
out.write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps({k: record[k] for k in ['proof_commit', 'tracked_project_files', 'selection', 'tools_changed', 'pushed', 'count_change']}, indent=2))
