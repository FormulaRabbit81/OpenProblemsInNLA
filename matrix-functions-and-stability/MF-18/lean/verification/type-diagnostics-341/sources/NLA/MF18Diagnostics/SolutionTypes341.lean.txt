import NLA.MF18.Solution
import Lean

private def diagnosticEraseBinderNames : Lean.Expr → Lean.Expr
  | .forallE _ t b info => .forallE .anonymous (diagnosticEraseBinderNames t) (diagnosticEraseBinderNames b) info
  | .lam _ t b info => .lam .anonymous (diagnosticEraseBinderNames t) (diagnosticEraseBinderNames b) info
  | .letE _ t v b nd => .letE .anonymous (diagnosticEraseBinderNames t) (diagnosticEraseBinderNames v) (diagnosticEraseBinderNames b) nd
  | .app f a => .app (diagnosticEraseBinderNames f) (diagnosticEraseBinderNames a)
  | .mdata m b => .mdata m (diagnosticEraseBinderNames b)
  | .proj n i b => .proj n i (diagnosticEraseBinderNames b)
  | e => e
open Lean in
run_cmd do
  let mut rows : Array Json := #[]
  for name in #[`NLA.MF18.certified_half, `NLA.MF18.stability_semantics, `NLA.MF18.pencil_evaluation, `NLA.MF18.pencil_degree_bound, `NLA.MF18.positive_average_and_sign, `NLA.MF18.homotopy_boundary_nonvanishing, `NLA.MF18.cayley_degree_and_count, `NLA.MF18.cayley_boundary_transfer, `NLA.MF18.monic_half_plane_count_homotopy, `NLA.MF18.bounded_degree_disk_count_homotopy, `NLA.MF18.regularized_root_count, `NLA.MF18.solution_factorization, `NLA.MF18.complementary_stability, `NLA.MF18.closed_disk_stability_limit, `NLA.MF18.limiting_equation_and_spectra, `NLA.MF18.reciprocal_count_identity, `NLA.MF18.selected_spectrum_count, `NLA.MF18.stein_identity, `NLA.MF18.simple_unit_root_pairing, `NLA.MF18.generalized_stein_pairing, `NLA.MF18.stable_space_dimension, `NLA.MF18.stable_space_in_kernel, `NLA.MF18.stein_rank_lower_bound, `NLA.MF18.full_complex_rank, `NLA.MF18.canonical_full_complex_rank] do
    let ci ← getConstInfo name
    rows := rows.push <| Json.mkObj [
      ("name", toJson name.toString),
      ("levelParams", toJson (ci.levelParams.map Name.toString)),
      ("type", toJson (reprStr ci.type)),
      ("binderNormalizedType", toJson (reprStr (diagnosticEraseBinderNames ci.type)))]
  IO.FS.writeFile "/private/tmp/nla-lean-next-20260915/MF18-type-diagnostics-341/solution-types.json" (Json.compress (toJson rows))

open Lean in
run_cmd do
  let mut rows : Array Json := #[]
  let mut pending : Array Name := #[`NLA.MF18.certified_half, `NLA.MF18.stability_semantics, `NLA.MF18.pencil_evaluation, `NLA.MF18.pencil_degree_bound, `NLA.MF18.positive_average_and_sign, `NLA.MF18.homotopy_boundary_nonvanishing, `NLA.MF18.cayley_degree_and_count, `NLA.MF18.cayley_boundary_transfer, `NLA.MF18.monic_half_plane_count_homotopy, `NLA.MF18.bounded_degree_disk_count_homotopy, `NLA.MF18.regularized_root_count, `NLA.MF18.solution_factorization, `NLA.MF18.complementary_stability, `NLA.MF18.closed_disk_stability_limit, `NLA.MF18.limiting_equation_and_spectra, `NLA.MF18.reciprocal_count_identity, `NLA.MF18.selected_spectrum_count, `NLA.MF18.stein_identity, `NLA.MF18.simple_unit_root_pairing, `NLA.MF18.generalized_stein_pairing, `NLA.MF18.stable_space_dimension, `NLA.MF18.stable_space_in_kernel, `NLA.MF18.stein_rank_lower_bound, `NLA.MF18.full_complex_rank, `NLA.MF18.canonical_full_complex_rank]
  let mut seen : Array Name := #[]
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.push name
    let ci ← getConstInfo name
    match ci.value? (allowOpaque := true) with
    | none => rows := rows.push <| Json.mkObj [
        ("name", toJson name.toString), ("hasBody", toJson false),
        ("bodyConstants", toJson (#[] : Array String))]
    | some body =>
      let cs := body.getUsedConstants
      rows := rows.push <| Json.mkObj [
        ("name", toJson name.toString), ("hasBody", toJson true),
        ("bodyConstants", toJson (cs.map Name.toString))]
      for c in cs do
        if c.toString.startsWith "NLA.MF18." || c.toString.startsWith "_private.NLA.MF18." then
          if !seen.contains c then pending := pending.push c
  IO.FS.writeFile "/private/tmp/nla-lean-next-20260915/MF18-type-diagnostics-341/certificate-body-route.json" (Json.compress (toJson rows))
