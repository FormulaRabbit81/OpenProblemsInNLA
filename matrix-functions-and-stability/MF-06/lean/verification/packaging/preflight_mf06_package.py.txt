checks = 0
def ck(value, msg):
    global checks
    checks += 1
    if not value:
        raise AssertionError(msg)

manifest = load(P / 'PACKAGE-MANIFEST.json')

email = re.compile(r'(?i)(?:george|g?stepaniants|sgstepaniants)[^\s@<>]{0,40}@[\w.-]+\.[A-Za-z]{2,}')
archive_files = 0
def privacy(data, name):
    global archive_files
    if name.endswith('.tar.gz'):
        with tarfile.open(fileobj=io.BytesIO(data), mode='r:gz') as archive:
            for member in archive.getmembers():
                ck(member.isfile() and not PurePosixPath(member.name).is_absolute() and '..' not in PurePosixPath(member.name).parts, 'ordinary safe archived file ' + name + ':' + member.name)
                archive_files += 1
                privacy(archive.extractfile(member).read(), member.name)
    elif name.endswith('.gz'):
        privacy(gzip.decompress(data), name[:-3])
    else:
        ck(email.search(data.decode('utf8', errors='ignore')) is None, 'George email absent: ' + name)
for rel, f in files.items():
    privacy(f.read_bytes(), rel)
