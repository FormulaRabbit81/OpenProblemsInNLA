def archive_review(src, label):
    src = Path(src)
    for name, suffix in [('REVIEW.md', '.md'), ('VERDICT.json', '-verdict.json'), ('MANIFEST.json', '-manifest.json')]:
        copy(src / name, 'reviews/' + label + suffix)
    dest = P / 'reviews' / (label + '-packet.tar.gz')
    with dest.open('wb') as raw, gzip.GzipFile(fileobj=raw, mode='wb', mtime=0) as gz, tarfile.open(fileobj=gz, mode='w') as tar:
        for f in sorted(src.rglob('*')):
            assert not f.is_symlink()
            if f.is_file():
                data = f.read_bytes()
                assert not re.search(r'(?i)(?:george|g?stepaniants|sgstepaniants)[^\s@<>]{0,40}@[\w.-]+\.[A-Za-z]{2,}', data.decode('utf8', errors='ignore')), f
                info = tarfile.TarInfo(src.name + '/' + str(f.relative_to(src)))
                info.size, info.mode = len(data), 0o644
                tar.addfile(info, io.BytesIO(data))
    return {'label': label, 'source_packet': str(src), 'manifest_sha256': sha(src / 'MANIFEST.json'),
            'archive': dest.name, 'archive_sha256': sha(dest), 'report': label + '.md'}
